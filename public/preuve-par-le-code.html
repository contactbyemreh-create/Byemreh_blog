import { useRef, useState } from "react";

export default function MagnetButton({ children, strength = 0.35 }) {
  const ref = useRef(null);
  const [pos, setPos] = useState({ x: 0, y: 0 });

  function handleMove(e) {
    const box = ref.current.getBoundingClientRect();
    const relX = e.clientX - box.left - box.width / 2;
    const relY = e.clientY - box.top - box.height / 2;
    setPos({ x: relX * strength, y: relY * strength });
  }

  return (
    <button
      ref={ref}
      onMouseMove={handleMove}
      onMouseLeave={() => setPos({ x: 0, y: 0 })}
      style={{ transform: `translate(${pos.x}px, ${pos.y}px)` }}
    >
      {children}
    </button>
  );
}

const field = createFlowField(canvas, { cell: 24 });
const particles = spawn(90, field.bounds);

function tick(t) {
  ctx.fillStyle = "rgba(11,12,16,0.08)";
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  for (const p of particles) {
    const angle = field.angleAt(p.x, p.y, t * 0.00012);
    p.vx += Math.cos(angle) * 0.05;
    p.vy += Math.sin(angle) * 0.05;
    p.x += p.vx *= 0.94;
    p.y += p.vy *= 0.94;
    draw(p, ctx); wrap(p, canvas);
  }
  requestAnimationFrame(tick);
}

import requests
from bs4 import BeautifulSoup

def audit(url: str) -> dict:
    html = requests.get(url, timeout=8).text
    soup = BeautifulSoup(html, "html.parser")

    title = soup.title.string if soup.title else ""
    meta = soup.find("meta", attrs={"name": "description"})
    images = soup.find_all("img")
    missing_alt = [i for i in images if not i.get("alt")]

    return {
        "title_len": len(title or ""),
        "has_meta_description": bool(meta),
        "images_missing_alt": len(missing_alt),
        "score": 100 - len(missing_alt) * 5,
    }
